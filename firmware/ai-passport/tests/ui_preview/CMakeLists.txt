cmake_minimum_required(VERSION 3.20)
project(scheme_b_preview C CXX ASM)
option(PREVIEW_DEVICE_HEAP "Exercise the device's 24 KiB LVGL heap" OFF)
if(PREVIEW_DEVICE_HEAP)
    add_compile_definitions(PREVIEW_DEVICE_HEAP=1)
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(APP "${CMAKE_CURRENT_SOURCE_DIR}/../../main")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../managed_components/lvgl__lvgl" lvgl)
file(GLOB FONTS CONFIGURE_DEPENDS "${APP}/fonts/netease_ui_font_*.c")
add_executable(scheme_b_preview main.c "${APP}/device_music_ui.c" "${APP}/device_music_ui_spec.c" ${FONTS})
target_include_directories(scheme_b_preview PRIVATE "${APP}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(scheme_b_preview PRIVATE lvgl m)
